#version 440

in vec4 vertexColour;
in vec3 pos;

out vec4 color;


vec4 grid(vec3 pos, float scale, float div, vec3 col) {
    vec2 coord = pos.xz * scale / 0.001;
    vec2 derivative = fwidth(coord);
    vec2 grid = abs(fract(coord - 0.5) - 0.5) / derivative;
    float line = min(grid.x, grid.y);
    float minimumz = min(derivative.y, 1);
    float minimumx = min(derivative.x, 1);

    vec4 myColor = vec4(col, 1.0 - min(line, 1.0));

	float gridSpacing = 0.01;
	float lineWidth   = 0.1;
	float distToLineX = abs(mod(pos.x, gridSpacing));
    float distToLineY = abs(mod(pos.y, gridSpacing));
	if (distToLineX < lineWidth || distToLineY < lineWidth)
    {
        myColor.r = 1.0;
    }

    myColor.ga *= div;
    return myColor;
}


void main()
{
	//color = vertexColour;

	color.rgba  = (grid(pos, 1, 1, vec3(0.6, 0.6, 0.6))).rgba;
    color.rgba += (grid(pos, 10, 0.2, vec3(0.005, 0.005, 0.005))).rgba;
}
